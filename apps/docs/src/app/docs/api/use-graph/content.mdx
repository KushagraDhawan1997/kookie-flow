title: useGraph
description: API reference for the useGraph hook
---

A hook for managing graph state with the controlled component pattern.

## Import

```tsx
import { useGraph } from '@kushagradhawan/kookie-flow';
```

## Usage

```tsx
function MyFlow() {
  const {
    entities,
    edges,
    onEntitiesChange,
    onEdgesChange,
    onConnect,
  } = useGraph({
    initialEntities: [
      { id: '1', type: 'default', position: { x: 0, y: 0 }, data: { label: 'Entity 1' } },
    ],
    initialEdges: [],
  });

  return (
    <KookieFlow
      entities={entities}
      edges={edges}
      onEntitiesChange={onEntitiesChange}
      onEdgesChange={onEdgesChange}
      onConnect={onConnect}
    />
  );
}
```

## Options

```tsx
interface UseGraphOptions {
  initialEntities?: Entity[];
  initialEdges?: Edge[];
}
```

| Option | Type | Default | Description |
|--------|------|---------|-------------|
| `initialEntities` | `Entity[]` | `[]` | Initial entities in the graph |
| `initialEdges` | `Edge[]` | `[]` | Initial edges connecting entities |

## Return Value

```tsx
interface UseGraphReturn {
  // State
  entities: Entity[];
  edges: Edge[];
  setEntities: React.Dispatch<React.SetStateAction<Entity[]>>;
  setEdges: React.Dispatch<React.SetStateAction<Edge[]>>;

  // Change handlers (pass to KookieFlow)
  onEntitiesChange: (changes: EntityChange[]) => void;
  onEdgesChange: (changes: EdgeChange[]) => void;
  onConnect: (connection: Connection) => void;

  // Convenience methods
  addEntity: (entity: Entity) => void;
  removeEntity: (id: string) => void;
  addEdge: (edge: Edge) => void;
  removeEdge: (id: string) => void;
  getEntity: (id: string) => Entity | undefined;
  getEdge: (id: string) => Edge | undefined;
  getConnectedEdges: (entityId: string) => Edge[];
}
```

### State

| Property | Type | Description |
|----------|------|-------------|
| `entities` | `Entity[]` | Current entities in the graph |
| `edges` | `Edge[]` | Current edges in the graph |
| `setEntities` | `Dispatch<SetStateAction<Entity[]>>` | React state setter for entities |
| `setEdges` | `Dispatch<SetStateAction<Edge[]>>` | React state setter for edges |

### Change Handlers

| Property | Type | Description |
|----------|------|-------------|
| `onEntitiesChange` | `(changes: EntityChange[]) => void` | Handles entity position, selection, add, remove, dimension, and data changes |
| `onEdgesChange` | `(changes: EdgeChange[]) => void` | Handles edge selection, add, remove changes |
| `onConnect` | `(connection: Connection) => void` | Creates an edge when sockets are connected |

### Convenience Methods

| Method | Type | Description |
|--------|------|-------------|
| `addEntity` | `(entity: Entity) => void` | Add a new entity |
| `removeEntity` | `(id: string) => void` | Remove an entity and its connected edges |
| `addEdge` | `(edge: Edge) => void` | Add a new edge |
| `removeEdge` | `(id: string) => void` | Remove an edge |
| `getEntity` | `(id: string) => Entity \| undefined` | Find an entity by ID |
| `getEdge` | `(id: string) => Edge \| undefined` | Find an edge by ID |
| `getConnectedEdges` | `(entityId: string) => Edge[]` | Get all edges connected to an entity |

## Examples

### Adding Entities Programmatically

```tsx
function FlowWithToolbar() {
  const { entities, edges, addEntity, onEntitiesChange, onEdgesChange, onConnect } = useGraph({
    initialEntities: [],
    initialEdges: [],
  });

  const handleAddEntity = () => {
    addEntity({
      id: `entity-${Date.now()}`,
      type: 'default',
      position: { x: Math.random() * 400, y: Math.random() * 400 },
      data: { label: 'New Entity' },
      inputs: [{ id: 'in', name: 'Input', type: 'any' }],
      outputs: [{ id: 'out', name: 'Output', type: 'any' }],
    });
  };

  return (
    <>
      <button onClick={handleAddEntity}>Add Entity</button>
      <KookieFlow
        entities={entities}
        edges={edges}
        onEntitiesChange={onEntitiesChange}
        onEdgesChange={onEdgesChange}
        onConnect={onConnect}
      />
    </>
  );
}
```

### Updating Entity Data

```tsx
function FlowWithDataUpdates() {
  const { entities, edges, setEntities, onEntitiesChange, onEdgesChange, onConnect } = useGraph({
    initialEntities,
    initialEdges,
  });

  const updateEntityLabel = (entityId: string, label: string) => {
    setEntities((entities) =>
      entities.map((entity) =>
        entity.id === entityId
          ? { ...entity, data: { ...entity.data, label } }
          : entity
      )
    );
  };

  return (
    <KookieFlow
      entities={entities}
      edges={edges}
      onEntitiesChange={onEntitiesChange}
      onEdgesChange={onEdgesChange}
      onConnect={onConnect}
      onEntityClick={(entity) => {
        const newLabel = prompt('New label:', entity.data.label);
        if (newLabel) updateEntityLabel(entity.id, newLabel);
      }}
    />
  );
}
```

### Deleting Selected Entities

```tsx
function FlowWithDelete() {
  const { entities, edges, setEntities, setEdges, onEntitiesChange, onEdgesChange, onConnect } = useGraph({
    initialEntities,
    initialEdges,
  });

  const deleteSelected = () => {
    const selectedIds = new Set(entities.filter((e) => e.selected).map((e) => e.id));

    // Remove selected entities
    setEntities((entities) => entities.filter((e) => !e.selected));

    // Remove edges connected to deleted entities
    setEdges((edges) =>
      edges.filter((e) => !selectedIds.has(e.source) && !selectedIds.has(e.target))
    );
  };

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Delete' || e.key === 'Backspace') {
        deleteSelected();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [deleteSelected]);

  return (
    <KookieFlow
      entities={entities}
      edges={edges}
      onEntitiesChange={onEntitiesChange}
      onEdgesChange={onEdgesChange}
      onConnect={onConnect}
    />
  );
}
```

### Finding Connected Entities

```tsx
function FlowWithHighlighting() {
  const { entities, edges, getConnectedEdges, getEntity } = useGraph({
    initialEntities,
    initialEdges,
  });

  const getConnectedEntities = (entityId: string): Entity[] => {
    const connectedEdges = getConnectedEdges(entityId);
    const connectedIds = new Set<string>();

    for (const edge of connectedEdges) {
      if (edge.source !== entityId) connectedIds.add(edge.source);
      if (edge.target !== entityId) connectedIds.add(edge.target);
    }

    return Array.from(connectedIds)
      .map(getEntity)
      .filter((e): e is Entity => e !== undefined);
  };

  // Use getConnectedEntities for highlighting, validation, etc.
}
```
